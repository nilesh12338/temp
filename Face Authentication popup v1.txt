<script>
  $(function() {
    var $popup = $('#faceAuthPopup');
    var $link = $('#popupCardLink');
    var isDragging = false;

    // Initialize jQuery UI Draggable
    $popup.draggable({
      axis: 'y',
      containment: '.rghtCol',
      distance: 5,
      cancel: '#closePopupBtn',
      start: function() {
        isDragging = true;
        $(this).css('cursor', 'grabbing');
      },
      stop: function() {
        $(this).css('cursor', 'grab');
        // Reset dragging state after a tiny delay so click event handler can check it
        setTimeout(function() {
          isDragging = false;
        }, 100);
      }
    });

    // Prevent link navigation if drag occurred
    $link.on('click', function(e) {
      if (isDragging) {
        e.preventDefault();
        e.stopPropagation();
      }
    });

    // Close button dismiss logic
    $('#closePopupBtn').on('click', function(e) {
      e.preventDefault();
      e.stopPropagation();

      $popup.addClass('slide-out');
      setTimeout(function() {
        $popup.hide();
      }, 400);
    });
  });
</script>
